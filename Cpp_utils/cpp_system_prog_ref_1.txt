This compact, single-page C++ System Programming Cheatsheet strips away boilerplate to provide high-density, production-ready code patterns for Linux/POSIX systems.

⚠️ Error Handling & Diagnostics
Always read errno immediately after a failing system call (-1 or MAP_FAILED).
cpp
#include <cstring>
#include <cerrno>
#include <iostream>

if (system_call() == -1) {
    std::cerr << "Error: " << std::strerror(errno) << " (Code: " << errno << ")\n";
}


📌 File I/O & Memory Mapping
cpp
#include <fcntl.h>
#include <unistd.h>
#include <sys/mman.h>
#include <sys/stat.h>

// Low-Level Direct IO
int fd = open("file.bin", O_RDWR | O_CREAT | O_TRUNC, 0644);
write(fd, data_ptr, size);
lseek(fd, 0, SEEK_SET); // Reset offset
read(fd, buf_ptr, size);
close(fd);

// Zero-Copy Memory Mapping (mmap)
struct stat sb;
fstat(fd, &sb);
void* addr = mmap(nullptr, sb.st_size, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
char* file_payload = static_cast<char*>(addr); // Access file directly like an array
munmap(addr, sb.st_size);


💡 RAII Resource Management (No Leaks)
Bind system handles to smart pointers or runtime stack frames to enforce deterministic cleanup.
cpp
#include <memory>
#include <unistd.h>

// Automatically close FDs on scope exit
auto fd_closer = [](int* fd) { if(fd && *fd >= 0) { ::close(*fd); delete fd; } };
using UniqueFD = std::unique_ptr<int, decltype(fd_closer)>;

// Exception-safe RAII Handle
class Handle {
    int fd;
public:
    Handle(const char* p, int f) : fd(open(p, f)) { if(fd < 0) throw std::runtime_error("err"); }
    ~Handle() { if(fd >= 0) close(fd); }
    Handle(const Handle&) = delete;             // Prevent duplicate close
    Handle& operator=(const Handle&) = delete;  // Prevent duplication
    Handle(Handle&& o) noexcept : fd(o.fd) { o.fd = -1; } // Transfer ownership
};

🌟 Processes, Signals & IPC
cpp
#include <sys/wait.h>
#include <csignal>
#include <sys/shm.h>

// Process Forking & Execution
if (pid_t pid = fork(); pid == 0) {
    char* args[] = {(char*)"/bin/ls", nullptr};
    execv(args[0], args); 
    _exit(1); // Failsafe
} else {
    int status; waitpid(pid, &status, 0);
}

// Async-Signal Safe Interruption Flag
volatile std::sig_atomic_t stop_flag = 0;
void handle(int sig) { stop_flag = 1; } // Keep handler minimal

// Shared Memory (IPC)
int shmid = shmget(ftok("keyfile", 65), 4096, 0666 | IPC_CREAT);
char* mem = (char*)shmat(shmid, nullptr, 0); // Attach
shmdt(mem);                                  // Detach
shmctl(shmid, IPC_RMID, nullptr);            // Destroy

📈 Non-Blocking Event Loop (epoll)
Scalable I/O multiplexing pattern for network and file descriptors.
cpp
#include <sys/epoll.h>

int epfd = epoll_create1(0);
struct epoll_event ev{}, events[10];
ev.events = EPOLLIN | EPOLLET; // Edge-triggered read
ev.data.fd = target_fd;
epoll_ctl(epfd, EPOLL_CTL_ADD, target_fd, &ev);

// Event loop
int nfds = epoll_wait(epfd, events, 10, -1);
for(int i = 0; i < nfds; ++i) {
    if(events[i].events & EPOLLIN) { /* Read from events[i].data.fd */ }
}

1. File I/O & System Calls
Low-level resource handling using POSIX API wrapped safely inside C++ primitives.
    • File Descriptor Handling (POSIX)
      cpp
      #include <fcntl.h>
      #include <unistd.h>
      #include <sys/types.h>
      
      // Open/Create a file
      int fd = open("log.txt", O_WRONLY | O_CREAT | O_TRUNC, 0644);
      if (fd == -1) { /* Handle error using errno */ }
      
      // Write buffer
      const char* msg = "System log entry\n";
      ssize_t bytes_written = write(fd, msg, strlen(msg));
      
      // Read buffer
      char buf[256];
      ssize_t bytes_read = read(fd, buf, sizeof(buf) - 1);
      
      close(fd);
      
    • Memory-Mapped Files (mmap)
      cpp
      #include <sys/mman.h>
      #include <sys/stat.h>
      
      int fd = open("data.bin", O_RDONLY);
      struct stat sb;
      fstat(fd, &sb);
      
      // Map file into process address space
      void* addr = mmap(nullptr, sb.st_size, PROT_READ, MAP_PRIVATE, fd, 0);
      if (addr == MAP_FAILED) { /* Error */ }
      
      // Read directly via pointer casting
      char* file_data = static_cast<char*>(addr);
      
      // Clean up
      munmap(addr, sb.st_size);
      close(fd);
      

2. RAII System Wrappers
To prevent leaks of system-level primitives (file descriptors, sockets, pointers), encapsulate them using Resource Acquisition Is Initialization (RAII) or modern smart pointers. [1, 2]
    • Modern Custom Deleter (std::unique_ptr)
      cpp
      #include <memory>
      #include <unistd.h>
      
      // Custom lambda deleter to close a file descriptor automatically
      auto fd_deleter = [](int* fd) {
          if (fd && *fd >= 0) {
              close(*fd);
              delete fd;
          }
      };
      
      using SafeFD = std::unique_ptr<int, decltype(fd_deleter)>;
      
    • Explicit RAII Wrapper Class
      cpp
      class FileDescriptor {
      public:
          explicit FileDescriptor(const char* path, int flags) {
              fd_ = open(path, flags);
              if (fd_ < 0) throw std::runtime_error("Failed to open file");
          }
      
          ~FileDescriptor() { // Deterministic cleanup
              if (fd_ >= 0) close(fd_); 
          }
      
          // Disable copy semantics to prevent double-freeing resources
          FileDescriptor(const FileDescriptor&) = delete;
          FileDescriptor& operator=(const FileDescriptor&) = delete;
      
          // Allow move semantics
          FileDescriptor(FileDescriptor&& other) noexcept : fd_(other.fd_) { other.fd_ = -1; }
          FileDescriptor& operator=(FileDescriptor&& other) noexcept {
              if (this != &other) {
                  if (fd_ >= 0) close(fd_);
                  fd_ = other.fd_;
                  other.fd_ = -1;
              }
              return *this;
          }
      
          int get() const { return fd_; }
      private:
          int fd_ = -1;
      };
      

3. Processes & Signals
System lifecycle management, process duplication, and handling interruptions.
    • Forking & Executing Processes
      cpp
      #include <sys/wait.h>
      #include <unistd.h>
      
      pid_t pid = fork();
      
      if (pid < 0) {
          // Fork failed
      } else if (pid == 0) {
          // Child Process
          char* args[] = {(char*)"/bin/ls", (char*)"-l", nullptr};
          execv(args[0], args);
          _exit(1); // Only reached if execv fails
      } else {
          // Parent Process
          int status;
          waitpid(pid, &status, 0); // Wait for child to finish
          if (WIFEXITED(status)) {
              int exit_code = WEXITSTATUS(status);
          }
      }
      
    • Signal Handling (Modern POSIX)
      cpp
      #include <csignal>
      
      // Use volatile sig_atomic_t for lock-free inter-thread/signal flags
      volatile std::sig_atomic_t signal_status = 0;
      
      void handle_signal(int signal) {
          signal_status = signal;
      }
      
      int main() {
          struct sigaction sa;
          sa.sa_handler = handle_signal;
          sigemptyset(&sa.sa_mask);
          sa.sa_flags = 0;
      
          // Intercept CTRL+C
          sigaction(SIGINT, &sa, nullptr);
      }
      

4. Concurrency & IPC
Native modern C++ multi-threading paired with essential system IPC constructs. [1, 2]
    • Threads & Condition Variables
      cpp
      #include <thread>
      #include <mutex>
      #include <condition_variable>
      
      std::mutex mtx;
      std::condition_variable cv;
      bool ready = false;
      
      void worker_thread() {
          std::unique_lock<std::mutex> lock(mtx);
          cv.wait(lock, []{ return ready; }); // Protects against spurious wakeups
          // Perform work...
      }
      
      void Controller() {
          {
              std::lock_guard<std::mutex> lock(mtx);
              ready = true;
          }
          cv.notify_one();
      }
      
    • Shared Memory (POSIX)
      cpp
      #include <sys/shm.h>
      #include <sys/ipc.h>
      
      // Generate unique key
      key_t key = ftok("shmfile", 65);
      
      // Create shared memory segment (1024 bytes)
      int shmid = shmget(key, 1024, 0666 | IPC_CREAT);
      
      // Attach to process memory layout
      char* str = (char*) shmat(shmid, nullptr, 0);
      
      // Detach and destroy segment
      shmdt(str);
      shmctl(shmid, IPC_RMID, nullptr);
