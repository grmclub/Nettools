## generate a python script to analyse FIX message statistics by order types , no. of incoming messages per session

Here is a standalone Python script using standard libraries to parse FIX logs (or raw FIX strings) and generate statistical breakdowns by Order Types (Tag 40 OrdType) and Incoming Messages per Session (Tag 49 SenderCompID / Tag 56 TargetCompID).
fix_stats_analyzer.py

#!/usr/bin/env python3
"""
FIX Message Statistics Analyzer
Extracts and summarizes:
1. Message counts per session (SenderCompID -> TargetCompID)
2. Order breakdown by Order Type (Tag 40: Market, Limit, Stop, etc.)
3. Message distribution by MsgType (Tag 35)
"""

from collections import Counter, defaultdict
import re
import sys

# Standard FIX Enum Mappings
ORD_TYPES = {
    "1": "Market",
    "2": "Limit",
    "3": "Stop / Stop Loss",
    "4": "Stop Limit",
    "1": "Market",
    "P": "Pegged",
}

MSG_TYPES = {
    "0": "Heartbeat (0)",
    "1": "Test Request (1)",
    "2": "Resend Request (2)",
    "3": "Reject (3)",
    "4": "Sequence Reset (4)",
    "5": "Logout (5)",
    "8": "Execution Report (8)",
    "A": "Logon (A)",
    "D": "New Order Single (D)",
    "F": "Order Cancel Request (F)",
    "G": "Order Cancel/Replace Request (G)",
    "j": "Business Message Reject (j)",
}

def parse_fix_tags(line):
    """Normalizes SOH/Pipe delimiters and extracts key-value pairs."""
    clean_line = line.strip().replace('\x01', '|').strip('|')
    tags = {}
    for item in clean_line.split('|'):
        if '=' in item:
            k, v = item.split('=', 1)
            tags[k] = v
    return tags

def analyze_fix_stats(file_path):
    session_counts = Counter()
    ord_type_counts = Counter()
    msg_type_counts = Counter()
    total_messages = 0

    with open(file_path, 'r', encoding='utf-8', errors='ignore') as f:
        for line in f:
            if "8=FIX" not in line:
                continue

            tags = parse_fix_tags(line)
            if not tags:
                continue

            total_messages += 1

            # 1. Track Session (SenderCompID -> TargetCompID)
            sender = tags.get('49', 'UNKNOWN_SENDER')
            target = tags.get('56', 'UNKNOWN_TARGET')
            session_key = f"{sender} -> {target}"
            session_counts[session_key] += 1

            # 2. Track Message Type (Tag 35)
            msg_type = tags.get('35', 'Unknown')
            msg_type_desc = MSG_TYPES.get(msg_type, f"Other ({msg_type})")
            msg_type_counts[msg_type_desc] += 1

            # 3. Track Order Types for order-related messages (New Orders, Replaces)
            if msg_type in ['D', 'G']:
                ord_type = tags.get('40', 'Unspecified')
                ord_type_desc = ORD_TYPES.get(ord_type, f"Other/Custom ({ord_type})")
                ord_type_counts[ord_type_desc] += 1

    # Output Results
    print("==================================================================")
    print(f"                FIX LOG STATISTICS SUMMARY                        ")
    print("==================================================================")
    print(f"Total FIX Messages Parsed: {total_messages}\n")

    print("--- 1. INCOMING MESSAGES PER SESSION ---")
    printf_fmt = "  %-35s : %-10d (%6.2f%%)"
    for session, count in session_counts.most_common():
        pct = (count / total_messages) * 100 if total_messages else 0
        print(printf_fmt % (session, count, pct))

    print("\n--- 2. BREAKDOWN BY ORDER TYPE (Tag 40) ---")
    total_orders = sum(ord_type_counts.values())
    if total_orders > 0:
        for ord_type, count in ord_type_counts.most_common():
            pct = (count / total_orders) * 100
            print(printf_fmt % (ord_type, count, pct))
    else:
        print("  No order messages (35=D/G) found in log.")

    print("\n--- 3. MESSAGE TYPE DISTRIBUTION (Tag 35) ---")
    for msg_desc, count in msg_type_counts.most_common():
        pct = (count / total_messages) * 100 if total_messages else 0
        print(printf_fmt % (msg_desc, count, pct))

    print("==================================================================")

if __name__ == "__main__":
    if len(sys.argv) < 2:
        print("Usage: python fix_stats_analyzer.py <path_to_fix_log>")
    else:
        analyze_fix_stats(sys.argv[1])

##generate a python script to analyse FIX messages and calculate order ack latencies and generate statistics by percentile rank 

Here is a standalone Python script using standard libraries to parse FIX logs, match orders with their corresponding execution acknowledgments (Tag 35=8 with OrdStatus 39=0 or 39=A), and compute latency percentiles ($p_{50}$, $p_{90}$, $p_{95}$, $p_{99}$).

fix_ack_latency_percentiles.py

#!/usr/bin/env python3
"""
FIX Order Acknowledgment Latency & Percentile Analyzer

Calculates latency between order submission (NewOrderSingle 35=D / CancelReplace 35=G)
and initial venue acknowledgment (ExecutionReport 35=8 with OrdStatus 39=0 or 39=A).

Outputs:
- Summary statistics (Count, Min, Mean, Max)
- Percentile distribution (p50, p90, p95, p99)
- Breakdown by SenderCompID / Session
"""

from datetime import datetime
import math
import sys


def parse_fix_time(ts_str):
    """Parses FIX UTCTimestamp (Tag 52 or 60): YYYYMMDD-HH:MM:SS[.mmm[uuu]]"""
    try:
        if '.' in ts_str:
            base, frac = ts_str.split('.', 1)
            # Standardize microseconds padding
            frac = (frac + '000000')[:6]
            clean_ts = f"{base}.{frac}"
            return datetime.strptime(clean_ts, "%Y%m%d-%H:%M:%S.%f")
        return datetime.strptime(ts_str, "%Y%m%d-%H:%M:%S")
    except ValueError:
        return None


def parse_fix_line(line):
    """Converts SOH (\x01) or pipe (|) delimited FIX string to a dict."""
    clean_line = line.strip().replace('\x01', '|').strip('|')
    tags = {}
    for item in clean_line.split('|'):
        if '=' in item:
            k, v = item.split('=', 1)
            tags[k] = v
    return tags


def calculate_percentile(data, percentile):
    """Calculates percentile using linear interpolation between closest ranks."""
    if not data:
        return 0.0
    sorted_data = sorted(data)
    n = len(sorted_data)
    if n == 1:
        return float(sorted_data[0])

    # Rank index
    rank = (percentile / 100.0) * (n - 1)
    lower = math.floor(rank)
    upper = math.ceil(rank)
    weight = rank - lower

    return sorted_data[lower] + (sorted_data[upper] - sorted_data[lower]) * weight


def analyze_latencies(file_path):
    # Tracking open requests: { ClOrdID: (SendingTime_datetime, SenderCompID) }
    pending_requests = {}
    latencies_all = []
    latencies_by_session = {}

    total_orders = 0
    total_acks = 0

    with open(file_path, 'r', encoding='utf-8', errors='ignore') as f:
        for line in f:
            if "8=FIX" not in line:
                continue

            tags = parse_fix_line(line)
            msg_type = tags.get('35')
            cl_ord_id = tags.get('11')
            snd_time_str = tags.get('52')
            sender = tags.get('49', 'UNKNOWN')

            if not (msg_type and cl_ord_id and snd_time_str):
                continue

            snd_time = parse_fix_time(snd_time_str)
            if not snd_time:
                continue

            # 1. New Order Single (35=D) or Cancel/Replace (35=G)
            if msg_type in ['D', 'G']:
                pending_requests[cl_ord_id] = (snd_time, sender)
                total_orders += 1

            # 2. Execution Report (35=8)
            elif msg_type == '8':
                ord_status = tags.get('39')
                exec_type = tags.get('150')

                # Filter for Acknowledged Statuses: 0=New, A=Pending New, E=Pending Replace
                is_ack = (ord_status in ['0', 'A']) or (exec_type in ['0', 'A'])

                if is_ack and cl_ord_id in pending_requests:
                    req_time, orig_sender = pending_requests.pop(cl_ord_id)
                    total_acks += 1

                    # Compute round-trip latency in milliseconds
                    delta_ms = (snd_time - req_time).total_seconds() * 1000.0

                    if delta_ms >= 0:  # Ignore out-of-order clock skewed anomalies
                        latencies_all.append(delta_ms)

                        if orig_sender not in latencies_by_session:
                            latencies_by_session[orig_sender] = []
                        latencies_by_session[orig_sender].append(delta_ms)

    # Output Diagnostics
    print("==========================================================================")
    print("           FIX ORDER ACKNOWLEDGMENT LATENCY PERCENTILE ANALYSIS           ")
    print("==========================================================================")
    print(f"Total New/Replace Orders Tracked : {total_orders}")
    print(f"Matched Acknowledgment Reports  : {total_acks}")

    if not latencies_all:
        print("\nNo matching Order-ACK latency pairs found.")
        return

    print_percentile_table("OVERALL SESSION LATENCY (ms)", latencies_all)

    # Per-Session Breakdown
    if len(latencies_by_session) > 1:
        print("\n" + "-" * 74)
        print("PER-SENDER BREAKDOWN")
        print("-" * 74)
        for sender, latencies in latencies_by_session.items():
            print_percentile_table(f"SenderCompID (49): {sender}", latencies)


def print_percentile_table(title, latencies):
    n = len(latencies)
    avg_lat = sum(latencies) / n
    min_lat = min(latencies)
    max_lat = max(latencies)

    p50 = calculate_percentile(latencies, 50)
    p90 = calculate_percentile(latencies, 90)
    p95 = calculate_percentile(latencies, 95)
    p99 = calculate_percentile(latencies, 99)

    print(f"\n>>> {title} (Sample Size: {n})")
    print(f"  Min   : {min_lat:10.3f} ms | Avg   : {avg_lat:10.3f} ms | Max   : {max_lat:10.3f} ms")
    print("  ------------------------------------------------------------------")
    print(f"  p50   : {p50:10.3f} ms (Median)")
    print(f"  p90   : {p90:10.3f} ms")
    print(f"  p95   : {p95:10.3f} ms")
    print(f"  p99   : {p99:10.3f} ms")


if __name__ == "__main__":
    if len(sys.argv) < 2:
        print("Usage: python fix_ack_latency_percentiles.py <path_to_fix_log>")
    else:
        analyze_latencies(sys.argv[1])

#!/usr/bin/env python3
"""
FIX Order Acknowledgment Latency & Percentile Analyzer

Calculates latency between order submission (NewOrderSingle 35=D / CancelReplace 35=G)
and initial venue acknowledgment (ExecutionReport 35=8 with OrdStatus 39=0 or 39=A).

Outputs:
- Console summary and percentile distribution
- CSV export for aggregated statistics
- CSV export for individual order latency records
"""

import csv
from datetime import datetime
import math
import sys


def parse_fix_time(ts_str):
    """Parses FIX UTCTimestamp (Tag 52 or 60): YYYYMMDD-HH:MM:SS[.mmm[uuu]]"""
    try:
        if '.' in ts_str:
            base, frac = ts_str.split('.', 1)
            frac = (frac + '000000')[:6]
            clean_ts = f"{base}.{frac}"
            return datetime.strptime(clean_ts, "%Y%m%d-%H:%M:%S.%f")
        return datetime.strptime(ts_str, "%Y%m%d-%H:%M:%S")
    except ValueError:
        return None


def parse_fix_line(line):
    """Converts SOH (\x01) or pipe (|) delimited FIX string to a dict."""
    clean_line = line.strip().replace('\x01', '|').strip('|')
    tags = {}
    for item in clean_line.split('|'):
        if '=' in item:
            k, v = item.split('=', 1)
            tags[k] = v
    return tags


def calculate_percentile(data, percentile):
    """Calculates percentile using linear interpolation between closest ranks."""
    if not data:
        return 0.0
    sorted_data = sorted(data)
    n = len(sorted_data)
    if n == 1:
        return float(sorted_data[0])

    rank = (percentile / 100.0) * (n - 1)
    lower = math.floor(rank)
    upper = math.ceil(rank)
    weight = rank - lower

    return sorted_data[lower] + (sorted_data[upper] - sorted_data[lower]) * weight


def compute_metrics(latencies):
    """Computes basic stats and percentiles for a list of latency values."""
    if not latencies:
        return {}
    n = len(latencies)
    return {
        "count": n,
        "min": min(latencies),
        "avg": sum(latencies) / n,
        "max": max(latencies),
        "p50": calculate_percentile(latencies, 50),
        "p90": calculate_percentile(latencies, 90),
        "p95": calculate_percentile(latencies, 95),
        "p99": calculate_percentile(latencies, 99),
    }


def analyze_and_export_latencies(file_path, stats_csv_path="fix_latency_stats.csv", records_csv_path="fix_latency_records.csv"):
    # Pending requests store: { ClOrdID: (SendingTime_datetime, SenderCompID, ReqTime_str) }
    pending_requests = {}
    
    # Detailed records for CSV export: list of dicts
    detailed_records = []
    
    latencies_all = []
    latencies_by_session = {}

    total_orders = 0
    total_acks = 0

    with open(file_path, 'r', encoding='utf-8', errors='ignore') as f:
        for line in f:
            if "8=FIX" not in line:
                continue

            tags = parse_fix_line(line)
            msg_type = tags.get('35')
            cl_ord_id = tags.get('11')
            snd_time_str = tags.get('52')
            sender = tags.get('49', 'UNKNOWN')

            if not (msg_type and cl_ord_id and snd_time_str):
                continue

            snd_time = parse_fix_time(snd_time_str)
            if not snd_time:
                continue

            # 1. New Order Single (35=D) or Cancel/Replace (35=G)
            if msg_type in ['D', 'G']:
                pending_requests[cl_ord_id] = (snd_time, sender, snd_time_str)
                total_orders += 1

            # 2. Execution Report (35=8)
            elif msg_type == '8':
                ord_status = tags.get('39')
                exec_type = tags.get('150')

                # Filter for Acknowledged Statuses: 0=New, A=Pending New, E=Pending Replace
                is_ack = (ord_status in ['0', 'A']) or (exec_type in ['0', 'A'])

                if is_ack and cl_ord_id in pending_requests:
                    req_time, orig_sender, req_time_str = pending_requests.pop(cl_ord_id)
                    total_acks += 1

                    # Compute round-trip latency in milliseconds
                    delta_ms = (snd_time - req_time).total_seconds() * 1000.0

                    if delta_ms >= 0:  # Ignore clock skew anomalies
                        latencies_all.append(delta_ms)

                        if orig_sender not in latencies_by_session:
                            latencies_by_session[orig_sender] = []
                        latencies_by_session[orig_sender].append(delta_ms)

                        detailed_records.append({
                            "ClOrdID": cl_ord_id,
                            "SenderCompID": orig_sender,
                            "MsgType": msg_type,
                            "OrderSendingTime": req_time_str,
                            "AckSendingTime": snd_time_str,
                            "LatencyMs": round(delta_ms, 3)
                        })

    # Output Console Diagnostics
    print("==========================================================================")
    print("           FIX ORDER ACKNOWLEDGMENT LATENCY PERCENTILE ANALYSIS           ")
    print("==========================================================================")
    print(f"Total New/Replace Orders Tracked : {total_orders}")
    print(f"Matched Acknowledgment Reports  : {total_acks}")

    if not latencies_all:
        print("\nNo matching Order-ACK latency pairs found.")
        return

    # Print Console Summaries
    print_percentile_table("OVERALL SESSION LATENCY (ms)", latencies_all)

    if len(latencies_by_session) > 1:
        print("\n" + "-" * 74)
        print("PER-SENDER BREAKDOWN")
        print("-" * 74)
        for sender, latencies in latencies_by_session.items():
            print_percentile_table(f"SenderCompID (49): {sender}", latencies)

    # -------------------------------------------------------------------------
    # CSV Export 1: Summary Statistics
    # -------------------------------------------------------------------------
    stats_rows = []
    
    # Overall row
    overall_metrics = compute_metrics(latencies_all)
    overall_metrics["Scope"] = "OVERALL"
    stats_rows.append(overall_metrics)

    # Per-session rows
    for sender, latencies in latencies_by_session.items():
        sess_metrics = compute_metrics(latencies)
        sess_metrics["Scope"] = f"Session: {sender}"
        stats_rows.append(sess_metrics)

    fieldnames_stats = ["Scope", "count", "min", "avg", "max", "p50", "p90", "p95", "p99"]

    with open(stats_csv_path, 'w', newline='', encoding='utf-8') as csvfile:
        writer = csv.DictWriter(csvfile, fieldnames=fieldnames_stats)
        writer.writeheader()
        for row in stats_rows:
            # Round float values for CSV formatting
            formatted_row = {
                k: (round(v, 3) if isinstance(v, float) else v)
                for k, v in row.items()
            }
            writer.writerow(formatted_row)

    print(f"\n[Exporter] Aggregated statistics saved to : {stats_csv_path}")

    # -------------------------------------------------------------------------
    # CSV Export 2: Individual Order Records
    # -------------------------------------------------------------------------
    fieldnames_records = ["ClOrdID", "SenderCompID", "MsgType", "OrderSendingTime", "AckSendingTime", "LatencyMs"]

    with open(records_csv_path, 'w', newline='', encoding='utf-8') as csvfile:
        writer = csv.DictWriter(csvfile, fieldnames=fieldnames_records)
        writer.writeheader()
        writer.writerows(detailed_records)

    print(f"[Exporter] Individual order records saved to: {records_csv_path}")
    print("==========================================================================")


def print_percentile_table(title, latencies):
    metrics = compute_metrics(latencies)
    print(f"\n>>> {title} (Sample Size: {metrics['count']})")
    print(f"  Min   : {metrics['min']:10.3f} ms | Avg   : {metrics['avg']:10.3f} ms | Max   : {metrics['max']:10.3f} ms")
    print("  ------------------------------------------------------------------")
    print(f"  p50   : {metrics['p50']:10.3f} ms (Median)")
    print(f"  p90   : {metrics['p90']:10.3f} ms")
    print(f"  p95   : {metrics['p95']:10.3f} ms")
    print(f"  p99   : {metrics['p99']:10.3f} ms")


if __name__ == "__main__":
    if len(sys.argv) < 2:
        print("Usage: python fix_ack_latency_percentiles.py <path_to_fix_log> [stats_csv_out] [records_csv_out]")
    else:
        file_in = sys.argv[1]
        stats_out = sys.argv[2] if len(sys.argv) > 2 else "fix_latency_stats.csv"
        records_out = sys.argv[3] if len(sys.argv) > 3 else "fix_latency_records.csv"
        
        analyze_and_export_latencies(file_in, stats_out, records_out)
