## build a PCAP network packet parser to diagnose live FIX traffic

Python PCAP FIX Analyzer (Powered by Scapy)

This production-grade script uses scapy to read .pcap or .pcapng files, reconstruct TCP payloads, split spliced FIX messages (when multiple FIX tags cross or share TCP segments), and extract core metrics like:
    TCP handshake and reset flags (SYN, FIN, RST)
    Sender / Target CompIDs (49, 56)
    Sequence numbers (34)
    Network-level packet arrival timestamps vs. FIX Application SendingTime (52)


#!/usr/bin/env python3
# fix_pcap_analyzer.py - PCAP Packet Parser for FIX Protocol Traffic

#usage: python fix_pcap_analyzer.py capture.pcap

import re
import sys
from datetime import datetime
from scapy.all import rdpcap, TCP, IP

# Common FIX MsgTypes
MSG_TYPES = {
    "0": "Heartbeat", "1": "TestRequest", "2": "ResendRequest", 
    "3": "Reject", "4": "SequenceReset", "5": "Logout", 
    "8": "ExecutionReport", "A": "Logon", "D": "NewOrderSingle", "F": "OrderCancelRequest"
}

def parse_fix_payload(payload_bytes):
    """
    Decodes raw TCP byte stream and splits multiple FIX messages in a single packet.
    Handles standard SOH (\x01) and pipe (|) delimiters.
    """
    try:
        # Decode ASCII bytes, replacing invalid byte sequences safely
        raw_str = payload_bytes.decode('latin1', errors='ignore')
    except Exception:
        return []

    # FIX messages start with 8=FIX...
    if "8=FIX" not in raw_str:
        return []

    # Split payloads if multiple messages exist in one TCP segment
    potential_msgs = raw_str.split("8=FIX")
    fix_messages = []

    for msg_fragment in potential_msgs:
        if not msg_fragment:
            continue
        full_msg = "8=FIX" + msg_fragment
        clean_msg = full_msg.replace('\x01', '|')
        
        # Parse fields into key-value map
        fields = {}
        for item in clean_msg.split('|'):
            if '=' in item:
                k, v = item.split('=', 1)
                fields[k] = v
                
        if fields:
            fix_messages.append((clean_msg, fields))
            
    return fix_messages

def analyze_pcap(pcap_path, fix_port=None):
    print(f"Reading capture file: {pcap_path}...\n")
    try:
        packets = rdpcap(pcap_path)
    except Exception as e:
        print(f"Error reading PCAP file: {e}")
        return

    parsed_count = 0

    for i, pkt in enumerate(packets, start=1):
        if not pkt.haslayer(TCP) or not pkt.haslayer(IP):
            continue

        ip_layer = pkt[IP]
        tcp_layer = pkt[TCP]
        
        # Filter by port if specified
        if fix_port and (tcp_layer.sport != fix_port and tcp_layer.dport != fix_port):
            continue

        pcap_timestamp = datetime.fromtimestamp(float(pkt.time)).strftime('%Y%m%d-%H:%M:%S.%f')[:-3]
        src_ip = ip_layer.src
        dst_ip = ip_layer.dst
        src_port = tcp_layer.sport
        dst_port = tcp_layer.dport
        flags = tcp_layer.flags

        # Detect TCP Control Flags (Session Diagnostics)
        if flags.RST:
            print(f"[Packet #{i}] {pcap_timestamp} | {src_ip}:{src_port} -> {dst_ip}:{dst_port} | ⚠️ TCP RST (Connection Reset)")
            continue
        if flags.FIN:
            print(f"[Packet #{i}] {pcap_timestamp} | {src_ip}:{src_port} -> {dst_ip}:{dst_port} | 🔴 TCP FIN (Connection Closed)")
            continue

        # Extract Raw Payload
        raw_payload = bytes(tcp_layer.payload)
        if not raw_payload:
            continue

        # Parse FIX Protocol Messages inside TCP Segment
        messages = parse_fix_payload(raw_payload)
        
        for raw_txt, tags in messages:
            parsed_count += 1
            msg_type = tags.get('35', 'Unknown')
            msg_name = MSG_TYPES.get(msg_type, f"MsgType={msg_type}")
            seq_num = tags.get('34', 'N/A')
            sender = tags.get('49', 'N/A')
            target = tags.get('56', 'N/A')
            sending_time = tags.get('52', 'N/A')

            print(f"[Packet #{i}] PCAP Time: {pcap_timestamp}")
            print(f" ├─ Socket     : {src_ip}:{src_port} -> {dst_ip}:{dst_port}")
            print(f" ├─ FIX Header : {sender} -> {target} | Seq: {seq_num}")
            print(f" ├─ Type       : {msg_type} ({msg_name})")
            print(f" ├─ Tag 52 Time: {sending_time}")
            
            # Print Rejects or Errors explicitly
            if msg_type in ['3', 'j']:
                text = tags.get('58', 'No explanation')
                print(f" └─ ❌ REJECT DETECTED: {text}")
            else:
                print(f" └─ Raw Payload: {raw_txt[:80]}...")
            print("-" * 60)

    print(f"\nAnalysis Complete. Extracted {parsed_count} FIX messages from PCAP.")

if __name__ == "__main__":
    if len(sys.argv) < 2:
        print("Usage: python fix_pcap_analyzer.py <path_to_pcap> [optional_fix_port]")
        print("Example: python fix_pcap_analyzer.py capture.pcap 9821")
    else:
        port = int(sys.argv[2]) if len(sys.argv) > 2 else None
        analyze_pcap(sys.argv[1], port)

##build a PCAP network packet parser to diagnose FIX traffic processing dissected messages from tshark. Do not use external libraries like python 

#!/usr/bin/env bash
# fix_pcap_tshark.sh - PCAP FIX Parser using tshark & native awk

PCAP_FILE="$1"
FIX_PORT="${2:-0}"

if [ -z "$PCAP_FILE" ]; then
    echo "Usage: $0 <pcap_file> [optional_fix_port]"
    exit 1
fi

# Build display filter if port is supplied
FILTER="fix"
if [ "$FIX_PORT" -gt 0 ]; then
    FILTER="tcp.port == $FIX_PORT && fix"
fi

echo "================================================================================"
echo " Parsing PCAP: $PCAP_FILE | Filter: $FILTER"
echo "================================================================================"

tshark -r "$PCAP_FILE" -Y "$FILTER" -T fields \
    -e frame.number \
    -e frame.time_epoch \
    -e ip.src \
    -e tcp.srcport \
    -e ip.dst \
    -e tcp.dstport \
    -e fix.SenderCompID \
    -e fix.TargetCompID \
    -e fix.MsgType \
    -e fix.MsgSeqNum \
    -e fix.SendingTime \
    -e fix.Text \
    -E header=n -E separator=/t -E occurrence=f | \
awk -F'\t' '
BEGIN {
    # FIX MsgType lookup map
    msg_types["0"]="Heartbeat"; msg_types["1"]="TestRequest"; msg_types["2"]="ResendRequest";
    msg_types["3"]="Reject"; msg_types["4"]="SequenceReset"; msg_types["5"]="Logout";
    msg_types["8"]="ExecutionReport"; msg_types["A"]="Logon"; msg_types["D"]="NewOrderSingle";
    msg_types["F"]="OrderCancelRequest"; msg_types["j"]="BusinessReject";

    count = 0;
}
{
    frame = $1;
    pcap_epoch = $2;
    src_ip = $3; src_port = $4;
    dst_ip = $5; dst_port = $6;
    sender = ($7 != "") ? $7 : "N/A";
    target = ($8 != "") ? $8 : "N/A";
    mtype = $9;
    seq = ($10 != "") ? $10 : "N/A";
    snd_time = ($11 != "") ? $11 : "N/A";
    err_text = $12;

    mname = (mtype in msg_types) ? msg_types[mtype] : "MsgType=" mtype;

    count++;

    # Print Packet Header Summary
    printf "\n[Frame #%-6s] Epoch: %.6f\n", frame, pcap_epoch;
    printf " ├─ Socket     : %s:%s -> %s:%s\n", src_ip, src_port, dst_ip, dst_port;
    printf " ├─ FIX Session: %s -> %s | MsgSeqNum: %s\n", sender, target, seq;
    printf " ├─ Message    : %s (%s)\n", mtype, mname;
    printf " ├─ SendingTime: %s\n", snd_time;

    # Highlight Rejects
    if (mtype == "3" || mtype == "j") {
        printf " └─ ❌ REJECT DETECTED: %s\n", (err_text != "") ? err_text : "No Text (Tag 58)";
    } else {
        printf " └─ Status     : OK\n";
    }
}
END {
    print "\n================================================================================";
    printf " Analysis Complete. Extracted %d FIX messages.\n", count;
    print "================================================================================";
}
'

##Generate an AWK script that calculates the millisecond latency delta between PCAP packet epoch timestamp and FIX Tag 52 SendingTime timestamp. 
fix_wire_latency.awk

#!/usr/bin/awk -f
# fix_wire_latency.awk - Calculate delta between PCAP frame arrival & Tag 52 SendingTime
# Expects tab-separated input from tshark:
# Field 1: frame.number
# Field 2: frame.time_epoch (e.g. 1785484800.123456)
# Field 3: fix.MsgType
# Field 4: fix.MsgSeqNum
# Field 5: fix.SendingTime (e.g. 20260731-08:00:00.100)

BEGIN {
    FS = "\t";
    printf "%-8s %-8s %-18s %-25s %-25s %-12s\n", "FRAME", "SEQ", "MSG_TYPE", "PCAP_TIME (UTC)", "FIX_SENDING_TIME", "DELTA (ms)";
    print "---------------------------------------------------------------------------------------------------";
}

{
    frame    = $1;
    pcap_ep  = $2;
    msg_type = $3;
    seq_num  = $4;
    snd_time = $5;

    # Skip records missing essential timestamp data
    if (pcap_ep == "" || snd_time == "") next;

    # -------------------------------------------------------------------------
    # 1. Parse FIX Tag 52 Timestamp: YYYYMMDD-HH:MM:SS.mmm
    # -------------------------------------------------------------------------
    # Split date/time string from milliseconds
    split(snd_time, time_parts, ".");
    dt_str = time_parts[1];  # YYYYMMDD-HH:MM:SS
    ms_str = time_parts[2];  # mmm (or uuuuuu)

    # Extract components: YYYY MMDD HH MM SS
    match(dt_str, /^([0-9]{4})([0-9]{2})([0-9]{2})-([0-9]{2}):([0-9]{2}):([0-9]{2})$/, date_bits);
    
    if (RSTART == 0) next; # Invalid timestamp format

    year  = date_bits[1];
    month = date_bits[2];
    day   = date_bits[3];
    hour  = date_bits[4];
    min   = date_bits[5];
    sec   = date_bits[6];

    # Convert milliseconds fraction to decimal epoch addition
    # Pad or truncate ms_str to 3 digits
    if (length(ms_str) < 3) {
        while (length(ms_str) < 3) ms_str = ms_str "0";
    } else if (length(ms_str) > 3) {
        ms_str = substr(ms_str, 1, 3);
    }
    fix_ms_fraction = ms_str / 1000.0;

    # Format for gawk mktime(): "YYYY MM DD HH MM SS [DST]"
    # Force UTC time environment during execution for accurate epoch conversion
    spec = sprintf("%04d %02d %02d %02d %02d %02d 0", year, month, day, hour, min, sec);
    fix_epoch_sec = mktime(spec, 1); # Second argument '1' specifies UTC in gawk

    if (fix_epoch_sec == -1) next;

    fix_total_epoch = fix_epoch_sec + fix_ms_fraction;

    # -------------------------------------------------------------------------
    # 2. Calculate Latency Delta (PCAP Packet Time - FIX Tag 52)
    # -------------------------------------------------------------------------
    delta_ms = (pcap_ep - fix_total_epoch) * 1000.0;

    # -------------------------------------------------------------------------
    # 3. Format Output
    # -------------------------------------------------------------------------
    printf "%-8s %-8s %-18s %-25s %-25s %-+12.3f\n", 
           frame, seq_num, msg_type, strftime("%Y%m%d-%H:%M:%S", pcap_ep, 1) "." sprintf("%03d", (pcap_ep - int(pcap_ep))*1000), snd_time, delta_ms;

    count++;
    sum_delta += delta_ms;
}

END {
    if (count > 0) {
        print "---------------------------------------------------------------------------------------------------";
        printf "Processed %d messages | Average Wire Delta: %.3f ms\n", count, (sum_delta / count);
    }
}
