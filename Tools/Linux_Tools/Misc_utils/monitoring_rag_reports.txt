##Monitoring rag report from csv data -bash

awk -F',' '
BEGIN {
    print "<html><head><style>"
    print "table { border-collapse: collapse; font-family: sans-serif; }"
    print "th, td { border: 1px solid #ccc; padding: 8px; }"
    print "th { background: #eee; }"
    print ".green { background-color: #28a745; color: white; font-weight: bold; text-align: center; }"
    print ".red { background-color: #dc3545; color: white; font-weight: bold; text-align: center; }"
    print "</style></head><body><table>"
}
NR==1 {
    print "<tr>"
    for(i=1; i<=NF; i++) print "<th>" $i "</th>"
    print "</tr>"
}
NR>1 {
    print "<tr>"
    for(i=1; i<NF; i++) print "<td>" $i "</td>"
    
    # Clean up value (lowercase/trim)
    val = $NF
    gsub(/^[ \t]+|[ \t]+$/, "", val)
    cls = (tolower(val) == "green") ? "green" : "red"
    
    print "<td class=\"" cls "\">" $NF "</td>"
    print "</tr>"
}
END {
    print "</table></body></html>"
}' report.csv > report.html

##Monitoring rag report from csv data -python


import csv

csv_data = """Server,Metric,Status
Web-01,CPU Usage,Green
Web-02,Memory,Red
DB-01,Disk Space,Green
"""

rows = list(csv.reader(csv_data.strip().splitlines()))
header, body = rows[0], rows[1:]

html_output = """<!DOCTYPE html>
<html>
<head>
<style>
  table { border-collapse: collapse; font-family: Arial, sans-serif; width: 100%; max-width: 600px; }
  th, td { border: 1px solid #ddd; padding: 10px; text-align: left; }
  th { background-color: #f4f4f4; }
  .status-green { background-color: #d4edda; color: #155724; font-weight: bold; text-align: center; }
  .status-red { background-color: #f8d7da; color: #721c24; font-weight: bold; text-align: center; }
</style>
</head>
<body>
<table>
"""

# Header
html_output += "  <tr>" + "".join(f"<th>{cell}</th>" for cell in header) + "</tr>\n"

# Body rows
for row in body:
    html_output += "  <tr>\n"
    for i, cell in enumerate(row):
        # Check if it's the last column
        if i == len(row) - 1:
            status_class = "status-green" if cell.strip().lower() == "green" else "status-red"
            html_output += f'    <td class="{status_class}">{cell}</td>\n'
        else:
            html_output += f'    <td>{cell}</td>\n'
    html_output += "  </tr>\n"

html_output += "</table>\n</body>\n</html>"

print(html_output)

##Convert csv output to html for a monitoring rag report last field is red or green

#!/bin/bash

cat << 'EOF' > convert.sh
#!/bin/bash

# Input CSV file
input_file="${1:-report.csv}"

# Read header
IFS=',' read -r -a header < "$input_file"

cat << HTML
<!DOCTYPE html>
<html>
<head>
<style>
  table { border-collapse: collapse; font-family: Arial, sans-serif; width: 100%; max-width: 700px; }
  th, td { border: 1px solid #ddd; padding: 10px; text-align: left; }
  th { background-color: #f4f4f4; }
  .status-green { background-color: #d4edda; color: #155724; font-weight: bold; text-align: center; }
  .status-red { background-color: #f8d7da; color: #721c24; font-weight: bold; text-align: center; }
</style>
</head>
<body>
<table>
  <tr>
HTML

for col in "${header[@]}"; do
  echo "    <th>$col</th>"
done

echo "  </tr>"

# Process data rows
first_line=true
while IFS=',' read -r -a fields; do
  # Skip header row
  if $first_line; then
    first_line=false
    continue
  fi

  # Skip empty lines
  [ ${#fields[@]} -eq 0 ] && continue

  echo "  <tr>"
  last_idx=$((${#fields[@]} - 1))

  for i in "${!fields[@]}"; do
    val="${fields[$i]}"
    if [ "$i" -eq "$last_idx" ]; then
      # Strip carriage returns / whitespace
      clean_val=$(echo "$val" | tr -d '\r' | xargs)
      if [[ "${clean_val,,}" == "green" ]]; then
        echo "    <td class=\"status-green\">$val</td>"
      else
        echo "    <td class=\"status-red\">$val</td>"
      fi
    else
      echo "    <td>$val</td>"
    fi
  done
  echo "  </tr>"
done < "$input_file"

cat << HTML
</table>
</body>
</html>
HTML
EOF

chmod +x convert.sh
./convert.sh report.csv > report.html

##Send rag report as html mail


#!/bin/bash

input_file="${1:-report.csv}"
recipient="admin@example.com"

# Counters & status flags
red_count=0
green_count=0
has_red=false

# Temporary buffer for table rows while counting
table_rows=""

# Read CSV line by line
first_line=true
while IFS=',' read -r -a fields; do
  [ ${#fields[@]} -eq 0 ] && continue

  if $first_line; then
    # Header row
    header_html="  <tr>\n"
    for col in "${fields[@]}"; do
      header_html+="    <th>$col</th>\n"
    done
    header_html+="  </tr>\n"
    first_line=false
    continue
  fi

  table_rows+="  <tr>\n"
  last_idx=$((${#fields[@]} - 1))

  for i in "${!fields[@]}"; do
    val="${fields[$i]}"
    if [ "$i" -eq "$last_idx" ]; then
      clean_val=$(echo "$val" | tr -d '\r' | xargs)
      
      if [[ "${clean_val,,}" == "red" ]]; then
        ((red_count++))
        has_red=true
        table_rows+="    <td class=\"status-red\">$val</td>\n"
      else
        ((green_count++))
        table_rows+="    <td class=\"status-green\">$val</td>\n"
      fi
    else
      table_rows+="    <td>$val</td>\n"
    fi
  done
  table_rows+="  </tr>\n"
done < "$input_file"

total_count=$((red_count + green_count))
subject="CRITICAL: RAG Monitoring Alert - ${red_count} Red / ${green_count} Green ($(date +'%Y-%m-%d %H:%M'))"

# Construct complete HTML document with summary header
html_content=$(cat << HTML
<!DOCTYPE html>
<html>
<head>
<style>
  body { font-family: Arial, sans-serif; color: #333; }
  .summary-card {
    display: inline-block;
    padding: 12px 20px;
    margin-bottom: 20px;
    border-radius: 6px;
    border: 1px solid #ddd;
    background-color: #f9f9f9;
  }
  .badge {
    padding: 4px 10px;
    border-radius: 4px;
    font-weight: bold;
  }
  .badge-red { background-color: #f8d7da; color: #721c24; }
  .badge-green { background-color: #d4edda; color: #155724; }
  table { border-collapse: collapse; width: 100%; max-width: 700px; }
  th, td { border: 1px solid #ddd; padding: 10px; text-align: left; }
  th { background-color: #f4f4f4; }
  .status-green { background-color: #d4edda; color: #155724; font-weight: bold; text-align: center; }
  .status-red { background-color: #f8d7da; color: #721c24; font-weight: bold; text-align: center; }
</style>
</head>
<body>

<h2>⚠️ RAG Monitoring Alert Report</h2>

<div class="summary-card">
  <strong>Total Checks:</strong> ${total_count} &nbsp;|&nbsp;
  <span class="badge badge-red">Red: ${red_count}</span> &nbsp;
  <span class="badge badge-green">Green: ${green_count}</span>
</div>

<table>
$(echo -e "$header_html")
$(echo -e "$table_rows")
</table>

</body>
</html>
HTML
)

# Only send email if at least one Red status was detected
if [ "$has_red" = true ]; then
  echo "$html_content" | mailx \
    -a "Content-Type: text/html; charset=UTF-8" \
    -s "$subject" \
    "$recipient"

  echo "[ALERT] ${red_count} Red status(es) detected. Email sent to $recipient."
else
  echo "[OK] All ${green_count} checks Green. No email sent."
fi
