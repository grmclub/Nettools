📌 Low-Level File I/O & Memory Mapping
Bypassing standard high-level streams for unbuffered, direct OS calls and zero-copy performance.
python
import os, mmap

# Low-Level OS File Descriptors
fd = os.open("data.bin", os.O_RDWR | os.O_CREAT | os.O_TRUNC, 0o644)
os.write(fd, b"System payload data")
os.lseek(fd, 0, os.SEEK_SET) # Reset offset
data = os.read(fd, 100)
# os.close(fd) managed via mmap or explicit call

# Zero-Copy Memory Mapping (mmap)
# Maps a file descriptor directly into the process virtual address space
with mmap.mmap(fd, 0, access=mmap.ACCESS_WRITE) as mm:
    print(mm[0:6])       # Read slices directly from memory
    mm[0:6] = b"Kernel"  # In-place file modification
    mm.flush()           # Force flush changes back to disk
os.close(fd)

proc = subprocess.Popen(["ping", "-c", "3", "127.0.0.1"], stdout=subprocess.PIPE)
stdout, _ = proc.communicate() # Await termination without deadlocking

# Async Signal Handling (POSIX)
def handle_sigint(signum, frame):
    print("\n[Interrupt] Graceful shutdown...")
    sys.exit(0)

signal.signal(signal.SIGINT, handle_sigint) # Trap Ctrl+C


🌟 Concurrency: Bypassing the GIL (Multiprocessing)
Because Python threads suffer from the Global Interpreter Lock (GIL), true multi-core parallel systems development requires separate OS processes or shared memory.
python
from multiprocessing import Process, Value, Array

def worker(shared_num, shared_arr):
    shared_num.value = 3.14159                     # Thread/Process safe write
    for i in range(len(shared_arr)): shared_arr[i] *= 2

if __name__ == "__main__":
    num = Value('d', 0.0)                          # Shared double precision float
    arr = Array('i', [1, 2, 3, 4])                 # Shared integer array
    
    p = Process(target=worker, args=(num, arr))
    p.start()
    p.join()                                       # Await completion
    print(num.value, arr[:])


📊 Interacting with C Libraries & Hardware (ctypes)

import ctypes

libc = ctypes.CDLL("libc.so.6")

# Set up argument and return types for the native function

libc.pow.restype = ctypes.c_double

result = libc.pow(2.0, 10.0) # Calls native compiled C loop directly

# Allocate raw, mutable hardware byte buffers
raw_buffer = ctypes.create_string_buffer(1024)
print(len(raw_buffer)) # 1024 bytes ready for raw system operations


📈 Network & Multiplexing (select)
Efficient event loops tracking multiple socket connections simultaneously without thread bloat.
python
import socket, select

server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
server.setblocking(False) # Non-blocking mode
server.bind(('localhost', 8080))
server.listen(5)

inputs = [server] # Keep track of descriptors to monitor

# High-density poll loop
while inputs:
    # Blocks until at least one monitored socket is ready for IO
    readable, _, _ = select.select(inputs, [], [])
    for s in readable:
        if s is server:
            client, addr = s.accept()
            client.setblocking(False)
            inputs.append(client) # Register client socket
        else:
            data = s.recv(1024)
            if data: s.send(b"ACK: " + data)
            else: inputs.remove(s); s.close() # Connection dropped

writing Unix Daemons in pure Python?
To run a Python script as a true Unix Daemon, it must be completely detached from the controlling terminal, handle system signals cleanly, and manage its own lifecycle.
Here is the definitive guide and a production-ready template to write a Unix daemon in pure Python without external dependencies.

🏛️ The Daemonization Protocol (The Double-Fork)
To decouple a process from the terminal, a daemon must follow a strict POSIX sequence:
    1. First Fork: Spawn a child and let the parent die. This guarantees the process isn't a process group leader.
    2. os.setsid(): Create a new session. The process becomes the leader of a new session and a new process group, completely losing its controlling terminal.
    3. Second Fork: Fork again and let the first child die. This ensures the daemon can never acquire a controlling terminal again (since only session leaders can acquire one).
    4. Sanitize Environment: Change the working directory to root (/) so it doesn't block unmounting filesystems, reset the umask, and redirect standard streams (stdin, stdout, stderr) to /dev/null or log files.

📜 Production-Ready Daemon Template
Save this file as my_daemon.py. It includes standard Double-Fork implementation, PID file tracking (to prevent multiple instances running), and Signal Handling.
python
import os
import sys
import time
import signal
import atexit

PID_FILE = "/tmp/my_daemon.pid"
LOG_FILE = "/tmp/my_daemon.log"

def daemonize():
    """Implements the standard POSIX double-fork daemonization protocol."""
    # 1. First Fork
    try:
        pid = os.fork()
        if pid > 0:
            sys.exit(0) # Exit parent
    except OSError as e:
        sys.stderr.write(f"Fork #1 failed: {e}\n")
        sys.exit(1)

    # Decouple from parent environment
    os.chdir("/")
    os.setsid()
    os.umask(0)

    # 2. Second Fork
    try:
        pid = os.fork()
        if pid > 0:
            sys.exit(0) # Exit first child
    except OSError as e:
        sys.stderr.write(f"Fork #2 failed: {e}\n")
        sys.exit(1)

    # Redirect standard file descriptors to avoid terminal leaks
    sys.stdout.flush()
    sys.stderr.flush()
    
    si = open(os.devnull, 'r')
    so = open(LOG_FILE, 'a+')
    se = open(LOG_FILE, 'a+')

    os.dup2(si.fileno(), sys.stdin.fileno())
    os.dup2(so.fileno(), sys.stdout.fileno())
    os.dup2(se.fileno(), sys.stderr.fileno())

    # Write process ID to PID file
    pid = str(os.getpid())
    with open(PID_FILE, 'w') as f:
        f.write(f"{pid}\n")
    
    # Register cleanup on exit
    atexit.register(lambda: os.remove(PID_FILE))

def handle_signals(signum, frame):
    """Gracefully handles termination signals from the OS."""
    if signum in (signal.SIGTERM, signal.SIGINT):
        print(f"[{time.ctime()}] Stopping daemon gracefully...", flush=True)
        sys.exit(0)
    elif signum == signal.SIGHUP:
        print(f"[{time.ctime()}] Reloading configuration...", flush=True)
        # Add config reload logic here

def daemon_main_loop():
    """The actual payload execution loop of your background service."""
    print(f"[{time.ctime()}] Daemon started successfully with PID {os.getpid()}", flush=True)
    
    while True:
        # --- YOUR CORE BACKGROUND TASK GOES HERE ---
        print(f"[{time.ctime()}] Daemon heartbeat tick.", flush=True)
        time.sleep(10)
        # -------------------------------------------

if __name__ == "__main__":
    # Setup Signal Traps
    signal.signal(signal.SIGTERM, handle_signals)
    signal.signal(signal.SIGINT, handle_signals)
    signal.signal(signal.SIGHUP, handle_signals)

    # Simple CLI Controller
    if len(sys.argv) < 2:
        print(f"Usage: {sys.argv[0]} [start|stop]")
        sys.exit(1)

    action = sys.argv[1]

    if action == "start":
        if os.path.exists(PID_FILE):
            print("Daemon is already running.")
            sys.exit(1)
        daemonize()
        daemon_main_loop()

    elif action == "stop":
        if os.path.exists(PID_FILE):
            with open(PID_FILE, 'r') as f:
                pid = int(f.read().strip())
            try:
                os.kill(pid, signal.SIGTERM)
                print("Stopping daemon...")
            except ProcessLookupError:
                print("PID file exists but process is already dead. Cleaning up.")
                os.remove(PID_FILE)
        else:
            print("Daemon is not running.")
    else:
        print("Unknown command. Use 'start' or 'stop'.")

⚡ The Modern Alternative: systemd
While the double-fork protocol is an essential system programming pattern, modern Linux environments lean heavily on systemd to handle lifecycle management. If you manage your script with systemd, you can keep your Python code flat (no forks) and let the OS handle the daemon logic.
